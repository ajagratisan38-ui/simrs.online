2. SETUP AWAL SERVER
═══════════════════════════════════════════════════════════════════════════════

  # Login ke VPS via SSH
  ssh root@IP_VPS_KAMU

  # Update system
  apt update && apt upgrade -y

  # Buat user non-root (JANGAN jalankan app sebagai root!)
  adduser simrs
  usermod -aG sudo simrs

  # Setup SSH key untuk user baru (opsional tapi sangat direkomendasikan)
  su - simrs
  mkdir -p ~/.ssh
  chmod 700 ~/.ssh
  # Copy public key kamu ke ~/.ssh/authorized_keys

  # Disable root login & password auth (setelah SSH key berfungsi)
  sudo nano /etc/ssh/sshd_config
  # → PermitRootLogin no
  # → PasswordAuthentication no
  sudo systemctl restart sshd

  # Setup timezone Indonesia
  sudo timedatectl set-timezone Asia/Jakarta

  # Setup firewall
  sudo ufw allow OpenSSH
  sudo ufw allow 80/tcp
  sudo ufw allow 443/tcp
  sudo ufw enable

  ✗ node_modules/      — Di-install ulang via Docker
  ✗ .next/, dist/      — Di-build ulang via Docker
  ✗ .env               — Secrets lokal (JANGAN sampai ke-push!)
  ✗ .env.production    — Secrets production
  ✗ .cursor/           — IDE config
  ✗ uploads/           — File upload user
  ✗ sik.sql, dll       — File referensi development

  ── METODE A: GIT CLONE (REKOMENDASI) ────────────────────────────────

  Langkah di PC lokal (sekali saja):

  # 1. Buat repository di GitHub
  #    - Buka https://github.com/new
  #    - Nama repo: simrs (atau simrs-saas)
  #    - Visibility: PRIVATE (kode ini bukan open source)
  #    - Jangan centang "Initialize with README" (sudah ada)
  #    - Klik "Create repository"

  # 2. Push dari PC lokal ke GitHub
  #    Buka terminal di folder SIMRS:
  cd "C:\Users\windows 10\Downloads\SIMRS"
  git remote add origin https://github.com/USERNAME_KAMU/simrs.git
  git push -u origin main

  #    Jika diminta login, masukkan:
  #    - Username: username GitHub kamu
  #    - Password: Personal Access Token (bukan password biasa!)
  #      Buat PAT di: https://github.com/settings/tokens
  #      Centang scope: repo (full control)

  Langkah di VPS:

  # 1. Login ke VPS
  ssh simrs@IP_VPS

  # 2. Clone repository
  cd /home/simrs
  git clone https://github.com/USERNAME_KAMU/simrs.git
  cd simrs

  # 3. Buat file .env production (TIDAK ada di GitHub!)
  nano .env
  #    → Copy isi dari .env.production di PC lokal
  #    → Atau salin dari .env.example lalu edit

  # 4. Verifikasi
  ls -la
  # Harus ada: apps/ packages/ docker/ .env package.json dll

  ls apps/server/src/migrations/
  # Harus ada 6 file migration

  head -8 .env
  # Harus tampil NODE_ENV=production

  ── METODE B: SCP MANUAL (jika tidak mau pakai GitHub) ───────────────

  CATATAN: Ganti IP_VPS dengan IP server kamu.

  # Dari Git Bash / WSL di PC lokal:

  # 1. Buat archive (exclude file yang tidak perlu)
  cd "/c/Users/windows 10/Downloads/SIMRS"
  tar czf simrs-deploy.tar.gz \
    --exclude='node_modules' \
    --exclude='.next' \
    --exclude='dist' \
    --exclude='.env' \
    --exclude='.env.local' \
    --exclude='.env.production' \
    --exclude='.cursor' \
    --exclude='uploads' \
    --exclude='.turbo' \
    --exclude='coverage' \
    --exclude='*.tsbuildinfo' \
    --exclude='sik.sql' \
    --exclude='sik_tables.txt' \
    --exclude='sik_table_names.txt' \
    apps/ packages/ docker/ tools/ docs/ \

  # 2. Upload ke VPS
  scp simrs-deploy.tar.gz simrs@IP_VPS:/home/simrs/

  # 3. Extract di VPS
  ssh simrs@IP_VPS
  mkdir -p /home/simrs/simrs
  cd /home/simrs/simrs
  tar xzf ../simrs-deploy.tar.gz
  rm ../simrs-deploy.tar.gz

  # 4. Upload .env.production sebagai .env
  # Dari PC lokal (PowerShell):
  scp "C:\Users\windows 10\Downloads\SIMRS\.env.production" simrs@IP_VPS:/home/simrs/simrs/.env

  # 5. Verifikasi
  ssh simrs@IP_VPS
  cd /home/simrs/simrs
  ls -la

5. KONFIGURASI ENVIRONMENT
═══════════════════════════════════════════════════════════════════════════════

  File .env.production sudah di-upload sebagai .env di step sebelumnya.
  Sekarang generate & isi secrets yang belum diisi.

  ssh simrs@IP_VPS
  cd /home/simrs/simrs

  # ── Generate secrets (jalankan satu per satu, copy hasilnya) ───────

  # Generate 3 secret keys:
  openssl rand -hex 32    # → paste ke JWT_SECRET
  openssl rand -hex 32    # → paste ke JWT_REFRESH_SECRET
  openssl rand -hex 32    # → paste ke ENCRYPTION_KEY

  # Generate password DB & Redis:
  openssl rand -base64 24  # → paste ke POSTGRES_PASSWORD (juga di DATABASE_URL)
  openssl rand -base64 24  # → paste ke REDIS_PASSWORD (juga di REDIS_URL)

  # Generate VAPID keys (perlu Node.js, bisa jalankan di lokal dulu):
  npx web-push generate-vapid-keys
  # → paste Public Key ke VAPID_PUBLIC_KEY dan NEXT_PUBLIC_VAPID_PUBLIC_KEY
  # → paste Private Key ke VAPID_PRIVATE_KEY

  # ── Edit .env dengan nano ──────────────────────────────────────────

  nano .env

  # Ganti semua value yang bertuliskan "GANTI_" dengan hasil generate di atas
  # PENTING: password di POSTGRES_PASSWORD harus SAMA dengan yang di DATABASE_URL
  # PENTING: password di REDIS_PASSWORD harus SAMA dengan yang di REDIS_URL

  # ── Verifikasi .env sudah benar ────────────────────────────────────

  grep "GANTI" .env
  # Harus return kosong (tidak ada yang belum diganti)

  grep "NODE_ENV" .env
  # Harus return: NODE_ENV=production

7. SETUP NGINX REVERSE PROXY + SSL
═══════════════════════════════════════════════════════════════════════════════

  # ── Buat konfigurasi Nginx ─────────────────────────────────────────────

  sudo nano /etc/nginx/sites-available/simrs

  # ── Isi dengan konfigurasi berikut ─────────────────────────────────────

  # --- Awal file konfigurasi ---

  # Rate limiting
  limit_req_zone $binary_remote_addr zone=api:10m rate=30r/s;
  limit_req_zone $binary_remote_addr zone=login:10m rate=5r/m;

  # Upstream backends
  upstream simrs_api {
      server 127.0.0.1:4001;
      keepalive 32;
  }

  upstream simrs_web {
      server 127.0.0.1:3000;
      keepalive 32;
  }

  # Redirect HTTP → HTTPS
  server {
      listen 80;
      server_name simrs.online *.simrs.online;
      return 301 https://$host$request_uri;
  }

  # Frontend (main domain + tenant subdomains)
  server {
      listen 443 ssl http2;
      server_name simrs.online *.simrs.online;

      # SSL (akan diisi oleh Certbot)
      ssl_certificate /etc/letsencrypt/live/simrs.online/fullchain.pem;
      ssl_certificate_key /etc/letsencrypt/live/simrs.online/privkey.pem;

      # Security headers
      add_header X-Frame-Options "SAMEORIGIN" always;
      add_header X-Content-Type-Options "nosniff" always;
      add_header X-XSS-Protection "1; mode=block" always;
      add_header Referrer-Policy "strict-origin-when-cross-origin" always;
      add_header Strict-Transport-Security "max-age=31536000; includeSubDomains" always;

      # Gzip
      gzip on;
      gzip_types text/plain text/css application/json application/javascript text/xml;
      gzip_min_length 1000;

      # Client body size (untuk upload file)
      client_max_body_size 10M;

      # API proxy
      location /api/ {
          limit_req zone=api burst=20 nodelay;
          proxy_pass http://simrs_api;
          proxy_http_version 1.1;
          proxy_set_header Upgrade $http_upgrade;
          proxy_set_header Connection 'upgrade';
          proxy_set_header Host $host;
          proxy_set_header X-Real-IP $remote_addr;
          proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
          proxy_set_header X-Forwarded-Proto $scheme;
          proxy_cache_bypass $http_upgrade;
          proxy_read_timeout 300s;
          proxy_connect_timeout 60s;
      }

      # Login rate limit lebih ketat
      location /api/v1/auth/login {
          limit_req zone=login burst=3 nodelay;
          proxy_pass http://simrs_api;
          proxy_http_version 1.1;
          proxy_set_header Host $host;
          proxy_set_header X-Real-IP $remote_addr;
          proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
          proxy_set_header X-Forwarded-Proto $scheme;
      }

      # Upload files
      location /uploads/ {
          proxy_pass http://simrs_api;
          proxy_http_version 1.1;
          proxy_set_header Host $host;
          proxy_set_header X-Real-IP $remote_addr;
      }

      # Frontend
      location / {
          proxy_pass http://simrs_web;
          proxy_http_version 1.1;
          proxy_set_header Upgrade $http_upgrade;
          proxy_set_header Connection 'upgrade';
          proxy_set_header Host $host;
          proxy_set_header X-Real-IP $remote_addr;
          proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
          proxy_set_header X-Forwarded-Proto $scheme;
          proxy_cache_bypass $http_upgrade;
      }

      # Static assets caching
      location /_next/static/ {
          proxy_pass http://simrs_web;
          proxy_cache_valid 200 60d;
          add_header Cache-Control "public, max-age=5184000, immutable";
      }

      # Health check
      location /health {
          proxy_pass http://simrs_api/api/health;
          access_log off;
      }
  }

  # --- Akhir file konfigurasi ---

  # ── Aktifkan site ──────────────────────────────────────────────────────

  sudo ln -s /etc/nginx/sites-available/simrs /etc/nginx/sites-enabled/
  sudo rm /etc/nginx/sites-enabled/default

  # Test konfigurasi
  sudo nginx -t

  # Reload
  sudo systemctl reload nginx

  # ── Setup SSL dengan Certbot ───────────────────────────────────────────

  # Untuk single domain:
  sudo certbot --nginx -d simrs.online -d www.simrs.online

  # Untuk wildcard subdomain (butuh DNS challenge):
  sudo certbot certonly --manual --preferred-challenges dns \
    -d simrs.online -d "*.simrs.online"

  # Auto-renew (biasanya sudah otomatis via certbot timer)
  sudo systemctl enable certbot.timer
  sudo systemctl start certbot.timer

  # Test auto-renew
  sudo certbot renew --dry-run

8. DOMAIN & DNS
═══════════════════════════════════════════════════════════════════════════════

  Atur DNS records di provider domain kamu:

  Type  | Name       | Value           | TTL
  ──────|────────────|─────────────────|─────
  A     | @          | IP_VPS_KAMU     | 300
  A     | *          | IP_VPS_KAMU     | 300
  A     | api        | IP_VPS_KAMU     | 300
  A     | admin      | IP_VPS_KAMU     | 300
  CNAME | www        | simrs.online        | 300

  Catatan:
  - Record * (wildcard) penting untuk multi-tenancy (rs-demo.simrs.online, dll)
  - Record api untuk endpoint API jika ingin pisah subdomain
  - Tunggu propagasi DNS 5-30 menit sebelum setup SSL

11. BACKUP OTOMATIS
═══════════════════════════════════════════════════════════════════════════════

  # ── Setup backup script ────────────────────────────────────────────────

  mkdir -p /home/simrs/simrs/backups
  chmod +x /home/simrs/simrs/docker/backup.sh

  # Test backup manual
  cd /home/simrs/simrs
  ./docker/backup.sh manual

  # ── Setup cron untuk backup otomatis ───────────────────────────────────

  crontab -e

  # Tambahkan baris berikut:

  # Backup harian jam 02:00 WIB
  0 2 * * * cd /home/simrs/simrs && ./docker/backup.sh daily >> /var/log/simrs-backup.log 2>&1

  # Backup mingguan hari Minggu jam 03:00 WIB
  0 3 * * 0 cd /home/simrs/simrs && ./docker/backup.sh weekly >> /var/log/simrs-backup.log 2>&1

  # ── Restore dari backup ────────────────────────────────────────────────

  # Decompress:
  gunzip backups/simrs_db_daily_20261104_020000.sql.gz

  # Restore:
  docker exec -i simrs-postgres psql -U simrs -d simrs_db \
    < backups/simrs_db_daily_20261104_020000.sql

  # ATAU buat database baru dulu (lebih aman):
  docker exec simrs-postgres createdb -U simrs simrs_db_restore
  docker exec -i simrs-postgres psql -U simrs -d simrs_db_restore \
    < backups/simrs_db_daily_20261104_020000.sql

  # ── Backup ke remote (opsional) ────────────────────────────────────────

  # SCP ke server lain:
  scp backups/simrs_db_daily_*.sql.gz user@backup-server:/backups/

  # Atau gunakan rclone ke Google Drive / S3:
  # rclone copy backups/ gdrive:simrs-backups/

13. TROUBLESHOOTING
═══════════════════════════════════════════════════════════════════════════════

  ── Problem: Container tidak start ──────────────────────────────────────

  docker logs simrs-server --tail 100
  docker logs simrs-web --tail 100

  # Cek apakah port sudah dipakai:
  sudo netstat -tlnp | grep -E '3000|4001|5432|6379'

  ── Problem: Database connection refused ────────────────────────────────

  # Cek apakah postgres berjalan:
  docker exec simrs-postgres pg_isready -U simrs

  # Cek DATABASE_URL di .env apakah host sudah benar:
  # Di docker-compose → host = "postgres" (nama service)
  # Di bare metal    → host = "localhost"

  ── Problem: "Migration table already exists" ───────────────────────────

  # Ini normal jika migrasi sudah pernah jalan. Skip saja.
  # Jika perlu force:
  docker exec simrs-server sh -c "cd apps/server && npx knex migrate:status --knexfile knexfile.ts"

  ── Problem: Redis connection refused ───────────────────────────────────

  docker exec simrs-redis redis-cli ping
  # Jika pakai password:
  docker exec simrs-redis redis-cli -a PASSWORD_REDIS ping

  ── Problem: 502 Bad Gateway (Nginx) ───────────────────────────────────

  # Backend belum ready, tunggu beberapa detik lalu refresh
  # Cek apakah backend berjalan:
  curl http://localhost:4001/api/health
  curl http://localhost:3000

  # Cek error log nginx:
  sudo tail -f /var/log/nginx/error.log

  ── Problem: SSL certificate error ─────────────────────────────────────

  sudo certbot renew --force-renewal
  sudo systemctl reload nginx

  ── Problem: Disk penuh ────────────────────────────────────────────────

  # Cek apa yang makan disk:
  du -sh /var/lib/docker/
  du -sh /home/simrs/simrs/backups/

  # Bersihkan docker:
  docker system prune -af
  docker volume prune -f

  # Bersihkan backup lama:
  find /home/simrs/simrs/backups -name "*.sql.gz" -mtime +30 -delete

  ── Problem: Out of Memory ─────────────────────────────────────────────

  # Tambah swap:
  sudo fallocate -l 2G /swapfile
  sudo chmod 600 /swapfile
  sudo mkswap /swapfile
  sudo swapon /swapfile
  echo '/swapfile none swap sw 0 0' | sudo tee -a /etc/fstab
